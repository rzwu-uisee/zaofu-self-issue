~~~java
class Solution {
    public int[] twoSum(int[] nums, int target) {
        // Map定义了变量的“类型”和“能力边界”，它关注的是 “能做什么”。
        // HashMap创建了一个具体的“对象”和“实现方式”，它关注的是 “具体怎么做”
        // 不要写HashMap<Integer,Ineger> map = new HashMap<Integer,Ineger>();如果后面需求改变，那我HashMap<Integer,Ineger> map
        // 就不可用了
        Map<Integer,Integer> map = new HashMap<Integer,Integer>();
        // i++/++i在循环里都是一样的
        for(int i=0;i<nums.length;i++){
            // 判断是否包含值用map.containsKey(key)，返回布尔类型
            // 获取键对应的值map.get(key)
            if(map.containsKey(target-nums[i])){
                // 注意是map.get(target-nums[i])获取下标，而不是填(target-nums[i])
                return new int []{map.get(target-nums[i]),i};
            }
            map.put(nums[i],i);
        }
        return new int [0];
    }
}
~~~

创建 HashMap 的根本原因就是用空间换时间。我们使用了一点额外的内存空间（哈希表）来记录已经遍历过的数字(键)及其位置(值)，从而将每次查找“配对数”（target-num[i]）的时间从 O(N)（遍历一次数组）降低到了 O(1)（查询一次哈希表），使得整个算法的总时间复杂度从 O(N²) 优化到了 O(N)。

~~~java
class Solution {
    public List<List<String>> groupAnagrams(String[] strs) {
        Map<String,List<String>> map = new HashMap<>();
        // 注意增强for的语法
        for(String str:strs){
            // 字符串转数组str.toCharArray
            char[] chars = str.toCharArray();
            // 字符数组排序Array.sort(chars)
            Arrays.sort(chars);
            // 字符数组转字符串
            String str_afterSort = String.valueOf(chars);
            if(map.containsKey(str_afterSort)){
                map.get(str_afterSort).add(str);
            }else{
                List<String> list = new ArrayList<>();
                list.add(str);
                map.put(str_afterSort,list);
            }
        }
        // 获取map集合的所有值集合map.values()
        List<List<String>> resultList = new ArrayList<>(map.values());
        return resultList;
    }
}
~~~
思路就是：

1. new一个Map集合Map<String,List<String>> map，键值对为：排序后的字符串=异位词集合
2. 遍历字符串数组中每个输入的字符串
2.1 将字符串转为字符数组(.toCharArray)
2.2 将字符数组排序Arrays.sort并转回字符串String.valueOf(因为所有异位词排序后的字符串都是一样的)

2.2.1 判断Map集合中的键是否包含排序后的字符串map.containsKey，如果包含，则把当前遍历的字符串add在排序后的字符串(键)对应的异位词集合(值)中map.get(str_afterSort)
2.2.2 如果不包含，则新建一个异位词集合List<String> list，并put进Map集合中(键为排序后的字符串，值为当前遍历的字符串)

3. 遍历结束后，先新建一个存在异位词集合的集合，然后从Map集合中获取(.values)所有值集合，再add进新建的集合中

~~~java
class Solution {
    public int longestConsecutive(int[] nums) {
        // Set 查询“有没有某个数”超级快，时间复杂度是 O(1)
        // Set 自动去重（如果数组里有重复数字，也只留一个，不影响结果）
        // 比如 输入 [100,4,200,1,3,2]，输出{1,2,3,4,100,200}
        Set<Integer> num_set = new HashSet<Integer>();
        for (int num : nums) {
            // 注意是add不是put
            num_set.add(num);
        }

        int longestStreak = 0;

        for (int num : num_set) {
            // 只有当前num是起点才会进入到计算连续序列长度的判断里面
            if (!num_set.contains(num - 1)) {
                int currentNum = num;
                int currentStreak = 1;
                //注意while循环放在if的里面
                while (num_set.contains(currentNum + 1)) {
                    currentNum += 1;
                    currentStreak += 1;
                }
                // 跳出while后要马上更新最大长度
                longestStreak = Math.max(longestStreak, currentStreak);
            }
        }

        return longestStreak;
    }
}
~~~

~~~java
class Solution {
    public void moveZeroes(int[] nums) {
        int n = nums.length, left = 0, right = 0;
        // 如果 left < right，说明 left 位置现在是个 0
        while (right < n) {
            // left在right指针查到零元素时交换后才会加一
            if (nums[right] != 0) {
                swap(nums, left, right);
                left++;
            }
            // right指针不管查没查到零元素都会加一
            right++;
        }
    }

    public void swap(int[] nums, int left, int right) {
        int temp = nums[left];
        nums[left] = nums[right];
        nums[right] = temp;
    }
}
~~~

~~~java
public class Solution {
    public int maxArea(int[] height) {
        // 注意右指针不能初始化为0下标
        int l = 0, r = height.length - 1;
        int ans = 0;
        while (l < r) {
            int area = Math.min(height[l], height[r]) * (r - l);
            // 注意在这里获取最大面积
            ans = Math.max(ans, area);
            // 如果 左边柱子更矮或一样高，那么当前这个左柱子已经是瓶颈了
            // 即使把右边界再往左移动，高度最多也只能保持不变
            //（因为右边本来就更高或一样高），面积只会变小
            // 所以果断抛弃左边这根矮柱子，把左指针右移，希望找到更高的柱子来提高高度
            if (height[l] <= height[r]) {
                ++l;
            }
            else {
                --r;
            }
        }
        return ans;
    }
}
~~~

~~~java
class Solution {
    public List<List<Integer>> threeSum(int[] nums) {
        Arrays.sort(nums); // 先排序
        List<List<Integer>> res = new ArrayList<>();
        
        for (int i = 0; i < nums.length; i++) {
            // 优化：如果当前数已经 > 0，后面的数更大，不可能凑出 0，直接结束
            if (nums[i] > 0) break;
            
            // 跳过重复元素
            if (i > 0 && nums[i] == nums[i - 1]) continue;
            
            // 双指针，目标是找到 nums[l] + nums[r] = -nums[i]
            // 固定 nums[i] 后，目标变成：在 i+1 ~ 末尾 找两个数 l 和 r，使得
            // nums[l] + nums[r] = -nums[i]
            int l = i + 1, r = nums.length - 1;
            int target = -nums[i];
            
            while (l < r) {
                int sum = nums[l] + nums[r];
                if (sum == target) {
                    res.add(Arrays.asList(nums[i], nums[l], nums[r]));
                    l++;
                    r--;
                    // 跳过重复元素
                    // 因为l\r已经移动了
                    while (l < r && nums[l] == nums[l - 1]) l++;
                    while (l < r && nums[r] == nums[r + 1]) r--;
                } else if (sum < target) {
                    l++;
                } else {
                    r--;
                }
            }
        }
        
        return res;
    }
}
~~~

~~~java
class Solution {
    public int lengthOfLongestSubstring(String s) {
        // 不停的判断是否窗口中有不重复的元素，如果不包含则扩大窗口，如果包含则把左边的元素移除，然后继续判断
        // 集合是窗口。i是左指针。right是右指针。
        // 哈希集合，记录每个字符是否出现过(去重)
        Set<Character> set = new HashSet<Character>();
        int n = s.length();
        // 右指针，初始值为 -1，相当于我们在字符串的左边界的左侧，还没有开始移动
        
        int right = -1, LongestSubstring = 0;
        for (int i = 0; i < n; ++i) {
            if (i != 0) {
                // 左指针向右移动一格，移除一个字符
                set.remove(s.charAt(i - 1));
            }
            // right + 1表示右指针目前指向的字符的下一个位置
            while (right + 1 < n && !set.contains(s.charAt(right + 1))) {
                // 不断地移动右指针
                set.add(s.charAt(right + 1));
                ++right;
            }
            // 第 i 到 right 个字符是一个极长的无重复字符子串
            // 窗口是从i-right，包含
            LongestSubstring = Math.max(LongestSubstring, right - i + 1);
            // set.size() == right - i + 1;
        }
        return LongestSubstring;
    }
}
~~~

~~~java
class Solution {
    // 把要匹配（短的）那个当窗口
    public List<Integer> findAnagrams(String s, String p) {
        List<Integer> res = new ArrayList<>();
        if (s.length() < p.length()) return res;

        int[] count = new int[26];
        // p 的频率
        for (char c : p.toCharArray()) {
            count[c - 'a']++;
        }
        // right 是窗口右边界，从 0 开始往右走
        // left 是左边界，一开始也在 0

        int left = 0;
        for (int right = 0; right < s.length(); right++) {
            // 右边进来的字符，消耗一次
            count[s.charAt(right) - 'a']--;

            // 如果窗口过大，或者刚刚那个字符已经负了 → 收缩左边
            if (count[s.charAt(right) - 'a'] < 0 || right - left + 1 > p.length()) {
                // 左边字符离开窗口，还回去
                count[s.charAt(left) - 'a']++;
                left++;
            }

            // 窗口大小正好 == p.length() 时检查
            if (right - left + 1 == p.length()) {
                // 只要没有负数，就说明完全匹配（因为总字符数已经相等）
                boolean match = true;
                for (int freq : count) {
                    if (freq < 0) {
                        match = false;
                        break;
                    }
                }
                if (match) {
                    res.add(left);
                }
            }
        }
        return res;
    }
}
~~~

~~~java
class Solution {
    public int subarraySum(int[] nums, int k) {
        // 键是前缀和，值是前缀和出现的次数
        Map<Integer, Integer> map = new HashMap<>();
        // 前缀和=0已出现了一次
        map.put(0, 1);
        // preSum 代表「从下标 0 到 当前位置」的元素和（包含当前位置）
        int preSum = 0;
        // 表示符合条件的子数组个数
        int res = 0;
        for (int i = 0; i < nums.length; i++) {
            preSum += nums[i];
            // 如果能找到前缀和为preSum[i] - k，那就代表已经找到和为k的连续子数组了
            if (map.containsKey(preSum - k)) {
                res += map.get(preSum - k);
            }
            // 把查到的前缀和存储起来
            map.put(preSum, map.getOrDefault(preSum, 0) + 1);
        }
        return res;
    }
}
~~~

~~~java
class Solution {
    public int maxSubArray(int[] nums) {
        // pre代表以当前元素结尾的连续子数组的最大和
        // maxAns用来存储找到的全局最大和，注意是初始化为数组第一个元素而不是0
        int pre = 0, maxAns = nums[0];
        for (int x : nums) {
            // 如果pre为正，那么pre + x 肯定比 x 大，所以应该把x拼接到pre
            // 如果pre为负，那么 pre + x 就会比 x 小，那么pre是个累赘，应该以x重新开始
            pre = Math.max(pre + x, x);
            maxAns = Math.max(maxAns, pre);
        }
        return maxAns;
    }
}
~~~

~~~java
class Solution {
    public int[][] merge(int[][] intervals) {
        if (intervals.length == 0) {
            // 返回二维空数组
            // 外层数组：长度为0的数组，里面没有任何元素
            // 内层数组：声明每个内层数组的长度为2，但因为外层数组长度为0，所以实际上没有任何内层数组被创建
            return new int[0][2];
        }
        // 根据区间的起点进行排序，谁小谁排前面
        Arrays.sort(intervals, new Comparator<int[]>() {
            public int compare(int[] interval1, int[] interval2) {
                
                return interval1[0] - interval2[0];
            }
        });
        // Arrays.sort(intervals, (interval1, interval2) -> interval1[0] - interval2[0]);
        
        List<int[]> merged = new ArrayList<int[]>();
        for (int i = 0; i < intervals.length; ++i) {
            int L = intervals[i][0], R = intervals[i][1];
            if (merged.size() == 0 || merged.get(merged.size() - 1)[1] < L) {
                merged.add(new int[]{L, R});
            } else {
                // 原本的区间里的左端点肯定更小
                // 合并后的区间右端点是两区间右端点的最大值
                merged.get(merged.size() - 1)[1] = Math.max(merged.get(merged.size() - 1)[1], R);
            }
        }
        return merged.toArray(new int[merged.size()][]);
    }
}
~~~

~~~java
class Solution {
    public void rotate(int[] nums, int k) {
        // 处理「k 可能比数组长度还大」的情况
        // 如果 k = 10 → 10 % 7 = 3 → 其实等价于旋转 3 步
        // 如果 k = 14 → 14 % 7 = 0 → 相当于不转
        k %= nums.length;
        // 整体翻转
        reverse(nums, 0, nums.length - 1);
        // 各自翻转
        reverse(nums, 0, k - 1);
        reverse(nums, k, nums.length - 1);
    }

    public void reverse(int[] nums, int start, int end) {
        while (start < end) {
            int temp = nums[start];
            nums[start] = nums[end];
            nums[end] = temp;
            start += 1;
            end -= 1;
        }
    }
}
~~~

~~~java
class Solution {
    public void setZeroes(int[][] matrix) {
        // 分别记录行长和列长
        int lengthOfRow = matrix.length;
        int lenthOfCollum = matrix[0].length;
        boolean CollumZero = false;
        boolean RowZero = false;
        // 检查第0列是否有0
        for(int i = 0;i<lenthOfCollum;i++){
            if(matrix[0][i]==0){
                RowZero = true;
            }
        }
        // 检查第0行是否有0
        for(int i = 0;i<lengthOfRow;i++){
            if(matrix[i][0]==0){
                CollumZero = true;
            }
        }
        // 用第0行和第0列“打标记”
        // 只扫描内部，不动第0行和第0列本身
        for(int i = 1;i<lengthOfRow;i++){
            for(int j = 1;j<lenthOfCollum;j++){
                if(matrix[i][j]==0){
                    // 发现0，就把这一行的第0列标记为0
                    // 发现0，就把这一列的第0列标记为0
                    matrix[i][0]=matrix[0][j]=0;
                }
            }
        }
        // 根据标记，把内部清零（从右下往左上清，避免干扰标记）        
        for(int i = 1;i<lengthOfRow;i++){
            for(int j = 1;j<lenthOfCollum;j++){
                // 如果这一行被标记了（matrix[i][0]==0）
                // 或者这一列被标记了（matrix[0][j]==0）
                // 就把当前位置清零
                if(matrix[i][0]==0||matrix[0][j]==0){
                    matrix[i][j]=0;
                }
            }
        }
        // 最后处理第0行和第0列（用之前记录的flag）       
        if(RowZero == true){
            for(int i = 0;i<lenthOfCollum;i++){
                // 更新第一列为0
                matrix[0][i]=0;
            }
        }
        if(CollumZero == true){
            for(int i = 0;i<lengthOfRow;i++){
                // 更新第一行为0
                matrix[i][0]=0;
            }
        }


    }
}
~~~

~~~java
class Solution {
    public List<Integer> spiralOrder(int[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;
        // 初始化上下左右四个方向的边界
        int upLimit = 0;
        int downLimit = row - 1;
        int leftLimt = 0;
        int rightLimit = col - 1;
        
        List<Integer> res = new ArrayList<>();

        while (true) {
            // 依次按照right -> down -> left -> up的顺序， 每个方向移动后边界值会变化
            // 终止条件为res中装了row * col个元素
            // 向右
            for (int j = leftLimt; j <= rightLimit; j++) {
                res.add(matrix[upLimit][j]);
            }
            upLimit++;
            if (res.size() == row * col) return res;
            // 向下
            for (int i = upLimit; i <= downLimit; i++) {
                res.add(matrix[i][rightLimit]);
            }
            rightLimit--;
            if (res.size() == row * col) return res;
            // 向左
            for (int j = rightLimit; j >= leftLimt; j--) {
                res.add(matrix[downLimit][j]);
            }
            downLimit--;
            if (res.size() == row * col) return res;
            // 向上
            for (int i = downLimit; i >= upLimit; i--) {
                res.add(matrix[i][leftLimt]);
            }
            leftLimt++;
            if (res.size() == row * col) return res;
        }
    }
}
~~~

~~~java
    public void rotate(int[][] matrix) {
        int n = matrix.length;
        for (int i = 0; i < n / 2; ++i) {
            for (int j = 0; j < (n + 1) / 2; ++j) {
                // [a][b] = [c][a] ，其中b+c = n-1
                int temp = matrix[i][j];
                matrix[i][j] = matrix[n - j - 1][i];
                matrix[n - j - 1][i] = matrix[n - i - 1][n - j - 1];
                matrix[n - i - 1][n - j - 1] = matrix[j][n - i - 1];
                matrix[j][n - i - 1] = temp;
            }
        }
    }
~~~

~~~java
class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        // m = 行数，n = 列数
        int m = matrix.length;          
        int n = matrix[0].length;       
        
        // 从右上角开始：第0行、最右列
        int x = 0;                      // 行指针，从第0行开始
        int y = n - 1;                  // 列指针，从最后一列开始
        
        // 只要没走出边界，就继续搜索
        while (x < m && y >= 0) {
            
            // 当前位置的值正好是 target → 找到了
            if (matrix[x][y] == target) {
                return true;
            }
            
            // 当前值比 target 大 → 目标在左边（或更上面，但上面已经排除了）
            // 所以向左移动一列
            if (matrix[x][y] > target) {
                --y;                    // y 减小，往左走
            } 
            // 当前值比 target 小 → 目标在下面（或更右边，但右边已经排除了）
            // 所以向下移动一行
            else {
                ++x;                    // x 增大，往下走
            }
        }
        
        // 走出边界了都没找到 → 不存在
        return false;
    }
}
~~~

~~~java
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }

public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        if (headA == null || headB == null) {
            return null;
        }
        // ListNode对象
        ListNode pA = headA, pB = headB;
        // 比较的是地址值
        while (pA != pB) {
            // 注意是headB/headA，而不是pB/pA
            pA = pA == null ? headB : pA.next;
            pB = pB == null ? headA : pB.next;
        }
        return pA;
    }
}
~~~

~~~java
    // 从右往左一位一位加，同时要处理进位（carry）
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        // 最终要返回的链表头
         ListNode head = null;
         // 用来不断在链表尾巴接新节点的指针
         // 永远指向当前链表的最后一个节点，这样我们每次想加新节点时，就能直接挂在 tail.next 上
         ListNode tail = null;
         // 进位
         int carry = 0;
         // 哪怕其中一个链表已经走完了，另一个还有，我们也要继续（用 0 来代替走完的部分）
         while(l1!=null || l2!=null){
            int n1 = l1==null?0:l1.val;
            int n2 = l2==null?0:l2.val;
            int sum = n1+n2+carry;
            // 第一次循环
            if(head==null){
                head = tail = new ListNode(sum%10);
            }else{
                // tail 后面接一个新节点
                // 并且移动tail指向新节点
                tail.next = new ListNode(sum%10);
                tail = tail.next;
            }
            carry = sum/10;
            if(l1!=null){
                l1 = l1.next;
            }
            if(l2!=null){
                l2 = l2.next;
            }
         }
         // 新进位
         // 如果最高位还有进位，就要在链表最后再加一个节点
         if(carry>0){
            tail.next = new ListNode(carry);
         }
         return head;
    }
}
~~~

~~~java
class Solution {
~~~

~~~java
    // 归并排序的核心思想：把大问题拆成小问题 → 解决小问题 → 合并小问题的结果
    // 自底向上版本的做法是：
    // 先把链表切成 1→1 的小段（长度=1）
    // 两两合并 → 得到很多长度=2 的有序小段
    // 再把长度=2 的小段两两合并 → 得到长度=4 的有序段
    // 再合并成 8、16、32…… 直到长度 ≥ 整个链表长度
    // 每次合并的子链表长度都是上一轮的 两倍
    public ListNode sortList(ListNode head) {
        if(head==null){
            return head;
        }
        // 计算总链表的长度
        int length = 0;
        ListNode node = head;
        while(node!=null){
            length++;
            node = node.next;
        }
        // 创建哑节点，方便统一处理头节点
        // 哑节点：方便在链表头部插入或修改时统一处理；最后返回 dummyHead.next 就是排序后的头
        // 为什么用哑节点？
        // 因为在合并过程中，头节点可能会变，用哑节点可以让代码写得更统一，最后直接返回 dummyHead.next 即可
        // 值为0，next指向头节点
        ListNode dummyHead = new ListNode(0,head);
        // 每一轮要合并的子链表长度翻倍
        for(int subLength =1;subLength<length;subLength*=2){
            // prev 指向已经排好序的部分的最后一个节点
            // curr 指向当前待处理的起点
            // 如果放for的外面去定义，会直接从链表中间开始拼接，而不是从头开始重新组织
            ListNode prev =dummyHead;
            ListNode curr = dummyHead.next;
            // 把整个链表分成多个长度为 subLength 的小段，两两合并
            while(curr!=null){
                // 先切第一个subLength 的子链表
                ListNode head1 =curr;
                // head1已经是第一个节点
                for(int i =1;i<subLength && curr.next !=null;i++){
                    curr = curr.next;
                }
                // 断开第一个子链表和后面的连接
                // 这里head2有可能为null,所以下面的for循环要判断curr!=null
                ListNode head2 = curr.next;
                curr.next =null;
                // curr 移动到第二个子链表的起点
                curr =head2;
                // 再切第二个子链表
                for(int i =1;i<subLength && curr!=null && curr.next!=null;i++){
                    curr = curr.next;
                }
                // 断开第二个子链表
                // next用来记住“第二段子链表切完之后，后面还剩下的链表起点
                ListNode next = null;
                // 因为curr有可能为null
                // curr.next==null也可以接受
                if(curr!=null){
                    next = curr.next;
                    curr.next =null;
                }
                // 合并两个有序子链表
                ListNode merged = merge(head1,head2);
                // 把合并后的结果接到已排序部分的后面
                prev.next = merged;
                // prev 移动到新合并段的末尾，为下一对合并做准备
                while(prev.next != null){
                    prev = prev.next;
                }
                // curr 跳到下一段要处理的起点
                curr = next;

            }
        }
        return dummyHead.next;
    }
    // 像拉链一样，一次比对两个链表的头结点，谁小就先拿谁
    // 谁被拿走了，就往后走一步

        // 可以不需要下面新节点，而是直接用head1,head2

        ListNode temp2 = head2;

                temp.next = temp1;
                // 移动而不是接上

            }else{
                temp.next = temp2;
                temp2 = temp2.next;

            temp = temp.next;

            temp.next =temp1;

            temp.next = temp2;
        }
~~~

~~~java
        int value;
        // prev：指向上一个节点
        // next：指向下一个节点
        // 因为是双向，所以我们可以很快地把一个节点从链表中间删掉（O(1)）
        Node prev;
        Node next;
        
        Node(int key, int value) {
            this.key = key;
            this.value = value;
        }
    }
    
    private final int capacity;              // 缓存容量
    private final Map<Integer, Node> map;    // key → 节点
    private final Node head;                 // 伪头节点（哨兵）
    private final Node tail;                 // 伪尾节点（哨兵）
    
    public LRUCache(int capacity) {
        this.capacity = capacity;
        this.map = new HashMap<>();
        
        // 初始化双向链表的两个哨兵节点
        // key = 0，value = 0
        head = new Node(0, 0);   //  dummy head
        tail = new Node(0, 0);   //  dummy tail
        // 双向链表
        head.next = tail;
        tail.prev = head;
    }
    
    public int get(int key) {
        Node node = map.get(key);
        if (node == null) {
            return -1;
        }
        // 命中 → 移到头部（最近使用）
        moveToHead(node);
        return node.value;
    }
    
    public void put(int key, int value) {
        Node node = map.get(key);
        
        // 情况1：key 已存在 → 更新值，并移到头部
        if (node != null) {
            node.value = value;
            moveToHead(node);
            return;
        }
        
        // 情况2：key 不存在 → 新建节点
        Node newNode = new Node(key, value);
        // 添加时要把key和node添加进去
        map.put(key, newNode);
        addToHead(newNode);   // 加到头部（最新使用）
        
        // 情况3：超出容量 → 删除尾部节点（最久未使用）
        if (map.size() > capacity) {
            // 伪尾节点的上一个就是真正的最后一个节点
            Node last = tail.prev;          // 真正的最后一个节点
            removeNode(last);
            // 移除时只需要移除key
            map.remove(last.key);
        }
    }
    
    // 把节点移动到链表头部（最近使用）
    private void moveToHead(Node node) {
        removeNode(node);     // 先从当前位置删除
        addToHead(node);      // 再插入到头部
    }
    
    // 把节点插入到头部（紧挨着 dummy head）
    private void addToHead(Node node) {
        node.prev = head;
        node.next = head.next;
        head.next.prev = node;
        head.next = node;
    }
    
    // 从链表中删除一个节点（前后指针连接）
    private void removeNode(Node node) {
        Node prev = node.prev;
        Node next = node.next;
        prev.next = next;
        next.prev = prev;
    }
}

/**
~~~
~~~java
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
~~~

~~~java
 *         this.right = right;
 *     }
 * }
 */
// 我先把左子树最右边的节点的右指针指向我自己，假装它是我的后继；
// 等我左子树遍历完，再回来时看到这个指针，就知道该访问我自己了，
// 然后断开这条线，继续往右走
class Solution {
    public List<Integer> inorderTraversal(TreeNode root) {
        List<Integer> res = new ArrayList<Integer>();
        // 前置节点：在当前节点有左子树时，找到左子树中最右边的那个节点
        //（也就是左子树里最后一个被访问的节点） 
        TreeNode predecessor = null;

        while (root != null) {
            // 先进入左树
            if (root.left != null) {
                // predecessor 节点就是当前 root 节点向左走一步，然后一直向右走至无法走为止
                predecessor = root.left;
                while (predecessor.right != null && predecessor.right != root) {
                    // 一直向右走
                    predecessor = predecessor.right;
                }
                
                // 让 predecessor 的右指针指向 root，继续遍历左子树
                if (predecessor.right == null) {
                    // 把左子树最右边的节点的右指针，临时指向回当前 root。留一个回去的路标
                    // 打上路标后继续往左走
                    predecessor.right = root;
                    root = root.left;
                }
                // 第二次绕回 root 时，predecessor.right 才会等于 root
                // 说明左子树已经访问完了，我们需要断开链接
                else {
                    res.add(root.val);
                    predecessor.right = null;
                    // 去右子树
                    root = root.right;
                }
            }
            // 如果没有左孩子，则直接访问右孩子
            else {
                res.add(root.val);
                root = root.right;
            }
        }
        return res;
    }
}
~~~

~~~java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;

 *     TreeNode(int val) { this.val = val; }
~~~

~~~java
 // I. 按层打印： 题目要求的二叉树的 从上至下 打印（即按层打印），又称为二叉树的 广度优先搜索（BFS）。BFS 通常借助 队列 的先入先出特性来实现。

 // II. 每层打印到一行： 将本层全部节点打印到一行，并将下一层全部节点加入队列，以此类推，即可分为多行打印。

// 总体思路：
// 用队列做 BFS（广度优先遍历），一层一层处理节点
// 每处理完一层，就把这一层的所有节点值收集到一个列表里，再把这个列表加到答案里。
// 关键技巧：

// 每次循环开始前，先记录当前层的节点个数（queue.size()）
// 只处理这么多节点，剩下的都是下一层的
// 处理完当前层后，把这一层的数值列表加到结果中

class Solution {
    public List<List<Integer>> levelOrder(TreeNode root) {
        List<List<Integer>> ret = new ArrayList<List<Integer>>();
        // 注意特殊情况处理
        if (root == null) {
            return ret;
        }
        // 队列
        Queue<TreeNode> queue = new LinkedList<TreeNode>();
        queue.offer(root);
        // 判断条件是队列非空
        while (!queue.isEmpty()) {
            // 注意这两个的存放位置
            List<Integer> level = new ArrayList<Integer>();
            int currentLevelSize = queue.size();
            for (int i = 0; i < currentLevelSize; ++i) {
                TreeNode node = queue.poll();
                level.add(node.val);
                if (node.left != null) {
                    queue.offer(node.left);
                }
                if (node.right != null) {
                    queue.offer(node.right);
                }
            }
            ret.add(level);
        }
        
        return ret;
    }
}
~~~

~~~java
        root.left = helper(nums, left, mid - 1);
        root.right = helper(nums, mid + 1, right);
~~~

~~~java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
// 原理：中序遍历得到的合法BST全程都是 当前值 > 前一个值
// 先把当前节点和其左节点先push进栈，然后弹一个出来，与inorder比较，赋值给inorder,然后往弹出来的节点的右边走，尝试把该节点的右边先push进去然后把右边的左节点也push进栈，然后循环往复
// 中序遍历 + 比较递增
// 因为中序遍历得到的合法BST全程都是 当前值 > 前一个值2 → 3 → 4 → 5 → 7
// 先创建一个栈和一个上一次中序访问的节点值inorder（设置为负值且很小）
// 然后当前栈不为空或者节点为空，就一直入栈且节点左移，移完就pop出来判断是不是小于等于inorder，如果不是就更新inorder值然后往右子树走
class Solution {
    public boolean isValidBST(TreeNode root) {
        // 创建一个栈
        Deque<TreeNode> stack = new ArrayDeque<TreeNode>();
        // 用来记录「上一次中序访问的节点值」
        // 一开始设置成一个很小的负数值
        double inorder = -Double.MAX_VALUE;
        // 注意是||而不是&&
        // 栈和当前节点 root 两个条件只要有一个不为空，就说明还有节点没处理完，必须继续循环
        // 栈里还有节点（之前压进去还没弹出来的）
        // 当前 root 还有右子树可以往下走（刚弹出一个节点后，要去它的右子树）
        while (!stack.isEmpty() || root != null) {
            // 一直往左子树走，走到最左边的叶子节点
            while (root != null) {
                stack.push(root);
                root = root.left;
            }
            root = stack.pop();
              // 如果当前节点的值 ≤ 上一个访问过的节点的值
              // 说明中序遍历不是严格递增 → 不是合法BST
            if (root.val <= inorder) {
                return false;
            }
            // 把当前节点的值记下来，作为「上一个值」
            inorder = root.val;
            // 处理弹出节点的右子树
            root = root.right;
        }
        return true;
    }
}
~~~

~~~java
        List<Integer> res = new ArrayList<>();
        if (root == null) {

                // 说明这是这一层最后一个被弹出来的节点，

                if (cur.left != null) {
                    queue.offer(cur.left);

                // 把右孩子（如果有）放进队列（下一层）
                if (cur.right != null) {
                    queue.offer(cur.right);
                }
            }
        }
~~~

~~~java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;

 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }

 *         this.right = right;
~~~

~~~java
class Solution {
    int[] preorder;
    // dic 把中序的值 → 下标 映射，方便 O(1) 查找根在中序的位置
    Map<Integer, Integer> dic = new HashMap<>();
    public TreeNode buildTree(int[] preorder, int[] inorder) {
        this.preorder = preorder;
        for(int i = 0; i < inorder.length; i++)
            dic.put(inorder[i], i);
        // recur(前序根索引, 中序左边界, 中序右边界
        return recur(0, 0, inorder.length - 1);
    }
    // 第一个参数 root：当前子树在前序数组里的根节点下标
    // 第二个参数 left：当前子树在中序数组的左边界（包含）
    // 第三个参数 right：当前子树在中序数组的右边界（包含）    
    TreeNode recur(int root, int left, int right) {
        if (left > right) return null;                          // 递归终止
        TreeNode node = new TreeNode(preorder[root]);          // 建立根节点
        // 当前子树的根节点，在中序遍历数组里的下标位置
        int i = dic.get(preorder[root]);                       // 划分根节点、左子树、右子树
        node.left = recur(root + 1, left, i - 1);              // 开启左子树递归
        // i-1(左子树的有边界)-left(左子树的左边界)+1 -- 算出来左子树有多少个节点
        // 然后结果再+rootIndex+1表示算到左子树根再前序中的下标
        node.right = recur(root + i - left + 1, i + 1, right); // 开启右子树递归
        return node;                                           // 回溯返回根节点
    }
}
~~~

~~~java
class Solution {

    int res = 0;
    Map<Long, Integer> map = new HashMap<>();
    // map记录「从根到当前路径上」出现过的所有前缀和，以及出现次数
    // 注意节点的取值范围要使用Long

    public int pathSum(TreeNode root, int targetSum) {
        // 注意先在map中放入0，不然当前缀和正好等于 targerSum 时没法被计入 res。
        // 注意这里要做 int -> Long 的类型转换
        map.put(0L, 1);
        // 这里也要做 int -> Long 的类型转换
        // 从根开始递归，当前路径和先设为 0（因为还没加上根节点的值）
        traversal(root, targetSum, 0L);
        return res;
    }

    // 由于节点值的范围是Long，所以节点的和的类型也应该是Long
    // 一边沿着树往下走（前序遍历），一边实时维护“从根到当前节点路径上的所有前缀和”，并在每个节点检查是否存在“某段路径和 = targetSum”的子路径
    public void traversal(TreeNode node, int targetSum, Long sum) {
        // 参数sum表示前缀和（从根节点到当前节点的所有节点之和）
        if (node == null)
            return;
        // 把当前节点的值加到前缀和里
        sum += node.val;
        if (map.containsKey(sum - targetSum)) {
            res += map.get(sum - targetSum);
        }
        map.put(sum, map.getOrDefault(sum, 0) + 1);
        // sum不能是全局变量，而必须是递归函数的参数。
        // 否则在这里向左递归后，sum的值的变化会影响到接下来向右递归时sum的大小
        traversal(node.left, targetSum, sum);
        traversal(node.right, targetSum, sum);

        // 回溯。因为sum是递归函数参数，每个递归分支的sum之间互相独立，
        // 接下来要尝试其他路径，不能让这条路的前缀和干扰别的路
        // 所以不用再手动回溯sum的值，只需要手动回溯map就够了
        map.put(sum, map.get(sum) - 1);
    }
~~~

~~~java
/**

 * public class TreeNode {
 *     int val;
 *     TreeNode left;

class Solution {
    // 递归还是有点难理解的，但是好记忆，记忆理解他
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        // 基本情况：空节点，或者当前节点就是 p 或 q 中的一个，直接返回它
        if (root == null || root == p || root == q) {
            return root;
        }
        
        // 去左子树找 p 或 q
        // 如果 root.left 这棵子树里同时包含 p 和 q → left 就是 p 和 q 在左子树里的 LCA
        // 如果 root.left 这棵子树里只包含 p 或 q 中的一个 → left 就指向那个被找到的节点（p 或者 q）
        // 如果 root.left 这棵子树里既不包含 p 也不包含 q → left = null
        TreeNode left = lowestCommonAncestor(root.left, p, q);
        
        // 去右子树找 p 或 q
        TreeNode right = lowestCommonAncestor(root.right, p, q);
        
        // 情况1：左右两边都找到了 → 当前节点就是 LCA
        if (left != null && right != null) {
            return root;
        }
        
        // 情况2：只有一个子树找到了 → 把找到的那个节点往上返回
        // 情况3：两个都没找到 → 返回 null（但在本题中一定存在，所以不会走到这里）
        return left != null ? left : right;
    }
}
~~~

~~~java
class Solution {
    public int numIslands(char[][] grid) {
        if (grid == null || grid.length == 0) return 0;
        
        int rows = grid.length;
        int cols = grid[0].length;
        int count = 0;
        
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (grid[i][j] == '1') {
                    count++;
                    // 从当前这个陆地格子 (i, j) 出发，去把所有和它相连的陆地都找到，并把它们都变成 '0'（沉入水下），这样我们就不会再重复数它们了
                    bfs(grid, i, j);
                }
            }
        }
        return count;
    }
    
    private void bfs(char[][] grid, int i, int j) {
        int rows = grid.length;
        int cols = grid[0].length;
        // 存放陆地坐标，队列
        Queue<int[]> queue = new LinkedList<>();
        queue.offer(new int[]{i, j});
        grid[i][j] = '0';  // 标记为已访问
        
        int[][] directions = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
        
        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            int x = cur[0];
            int y = cur[1];
            
            for (int[] dir : directions) {
                int nx = x + dir[0];
                int ny = y + dir[1];
                
                if (nx >= 0 && nx < rows && ny >= 0 && ny < cols && grid[nx][ny] == '1') {
                    queue.offer(new int[]{nx, ny});
                    grid[nx][ny] = '0';  // 标记
                }
            }
        }
    }
~~~

~~~java
class Solution {
    public int orangesRotting(int[][] grid) {
        if (grid == null || grid.length == 0) {
            return 0;
        }
        
        int m = grid.length;
        int n = grid[0].length;
        
        // 队列：存放当前这一分钟会腐烂的橘子坐标 {i, j}
        Queue<int[]> queue = new LinkedList<>();
        
        // 四个方向：右、左、下、上
        int[][] directions = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
        
        int freshCount = 0;  // 新鲜橘子（1）的数量
        
        // 第一遍遍历：统计新鲜橘子 + 把所有一开始就腐烂的橘子(2)入队
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 2) {
                    queue.offer(new int[]{i, j});
                } else if (grid[i][j] == 1) {
                    freshCount++;
                }
            }
        }
        
        // 特殊情况：一开始没有腐烂橘子，但有新鲜橘子 → 永远烂不了
        if (freshCount > 0 && queue.isEmpty()) {
            return -1;
        }
        
        int minutes = 0;  // 需要的分钟数
        
        // BFS 多源广度优先
        while (!queue.isEmpty()) {
            int size = queue.size();  // 当前这一分钟要处理的橘子数量
            
            // 处理当前层的所有腐烂橘子
            for (int k = 0; k < size; k++) {
                int[] cell = queue.poll();
                int x = cell[0];
                int y = cell[1];
                
                // 向四个方向尝试传染
                for (int[] dir : directions) {
                    // 计算邻居坐标
                    int nx = x + dir[0];
                    int ny = y + dir[1];
                    
                    // 合法边界 + 是新鲜橘子
                    if (nx >= 0 && nx < m && ny >= 0 && ny < n && grid[nx][ny] == 1) {
                        grid[nx][ny] = 2;           // 腐烂
                        queue.offer(new int[]{nx, ny});
                        freshCount--;               // 新鲜橘子减少
                    }
                }
            }
            
            // 如果这一轮有新的橘子被感染（队列不空），时间 +1
            if (!queue.isEmpty()) {
                minutes++;
            }
        }
        
        // 最后检查是否还有新鲜橘子没被感染
        return freshCount == 0 ? minutes : -1;
    }
}
~~~

~~~java
class Solution {
    // numCourses：总共有几门课
    // prerequisites：先修关系数组，每一项是 [a, b] → 想学 a 必须先学 b
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        // 用于记录每门课程的先修课程有多少门
        // a->b，表示a是先修，b是后修
        int[] inDegree = new int[numCourses];
        // 邻接表，存储每门课程的后续课程
        List<List<Integer>> adjList = new ArrayList<>();
        for (int i = 0; i < numCourses; i++) {
            // adjList的每一个元素表示学完 i 之后 可以去学 的所有课程
            adjList.add(new ArrayList<>());
        }

        // 计算每门课程的入度，并构建邻接表
        for (int[] prerequisite : prerequisites) {
            int course = prerequisite[0];
            int preCourse = prerequisite[1];
            // 
            inDegree[course]++;
            adjList.get(preCourse).add(course);
        }

        // 队列存储不需要先修课程的课程
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < numCourses; i++) {
            if (inDegree[i] == 0) {
                queue.offer(i);
            }
        }

        // 记录已完成课程的数量
        int count = 0;
        while (!queue.isEmpty()) {
            // 提取不需要先修课程的课程，因为不需要先修课程，可以直接完成
            int selectedCourse = queue.poll();
            count++;
            // 获取当前课程的后续课程列表
            List<Integer> nextCourses = adjList.get(selectedCourse);
            for (int nextCourse : nextCourses) {
                // 后续课程的入度减 1
                inDegree[nextCourse]--;
                if (inDegree[nextCourse] == 0) {
                    queue.offer(nextCourse);
                }
            }
        }

        // 如果已完成课程的数量等于总课程数，则可以完成所有课程
        return count == numCourses;
    }
~~~

~~~java
class Solution {
    // 解题思路：准备一个和输入数组长度一样长的flag布尔数组，用来标识数组里面的元素是否被使用过。再准备一个路径用来存储拿到的数组元素。然后利用回溯。
    // 当路径长度和数组长度 一样时，添加进答案列表。
    // 不一样时就遍历数组元素，判断当前元素是否被使用过，没被使用就添加进路径，并标注使用过，然后再继续回溯。
    // 回溯退出后要恢复现场。包括还原元素的标注和移除路径元素
    public List<List<Integer>> permute(int[] nums) {
        // 创建一个长度和 nums 相同的 boolean 数组
        // flags[i] = true 代表：nums[i] 这个数字已经被用过了
        boolean[] flags = new boolean[nums.length];
        List<List<Integer>> res = new ArrayList<>();
        // 空路径
        dfs(nums, res, flags, new ArrayList<>());
        return res;
    }


    public void dfs(int[] nums, List<List<Integer>> res, boolean[] flags, List<Integer> temp) {
        // 当我们已经选了和原数组一样多的数字 → 说明这条路径已经构成一个完整排列
        if (temp.size() == nums.length) {
            // 把当前这条路径（temp）拷贝一份加到答案里
            res.add(new ArrayList<>(temp));
            return;
        }
        for (int i = 0; i < nums.length; i++) {
            // 如果这个数字还没被用过
            if (!flags[i]) {
                flags[i] = true;
                temp.add(nums[i]);
                dfs(nums, res, flags, temp);
                
                // 恢复现场，注意是先构成一个完整的temp才会恢复现场
                // 把这个数字的“已使用”标记撤销掉，让它在别的分支里还能被选
~~~

~~~java
            // 把当前路径 path 完整拷贝一份 加到答案里
            res.add(new ArrayList<>(path));
            return;

            sum += nums[i];
            rollBack(nums, target, i, sum);
            // 恢复现场
            path.remove(path.size() - 1);
            sum -= nums[i];
        }
    }
~~~

~~~java
        // 因为搜索是从头开始匹配的，从稀有字母开始匹配往往能更快失败，剪枝效果更好
        if (cnt[word.charAt(word.length() - 1)] < cnt[word.charAt(0)]) {
            word = new StringBuilder(word).reverse().toString();
        }

                    return true;
                }
            }
        }
        return false;
    }

    private boolean dfs(char[][] board, String word, int i, int j, int k) {

        if (board[i][j] != word.charAt(k)) {
            return false;
        }
        // 如果当前已经匹配到 word 的最后一个字符 → 成功
        if (k == word.length() - 1) {
            return true;
        }
        
        // 先把当前格子的字母保存起来（temp）
        // 然后把这个格子“打上标记”（这里用 '\0' 表示已访问）
        char temp = board[i][j];
        board[i][j] = '\0';
        
        int m = board.length;
        int n = board[0].length;
        // 左、右、上、下
        int[][] directions = {{0, -1}, {0, 1}, {-1, 0}, {1, 0}};
        
        for (int[] dir : directions) {
            // 尝试四个方向的相邻格子坐标 (x,y)

                return true;
            }
        }
        
        // 恢复现场
        board[i][j] = temp;
        return false;
~~~

~~~java
class Solution {
    public List<List<String>> partition(String s) {
        List<List<String>> result = new ArrayList<>();
        // 用来保存当前正在构建的一条分割路径
        List<String> path = new ArrayList<>();
        
        backtrack(s, 0, path, result);
        return result;
    }
    
    private void backtrack(String s, int start, List<String> path, List<List<String>> result) {
        // 终止条件：已经处理到字符串末尾
        // 说明当前 path 是一条完整的分割方案
        if (start == s.length()) {
            // 必须 new 一个新列表，不然所有路径指向同一个对象
            result.add(new ArrayList<>(path));
            return;
        }
        
        // 从 start 开始，尝试各种结束位置 end
        for (int end = start; end < s.length(); end++) {
            // 如果 [start, end] 这段是回文
            if (isPalindrome(s, start, end)) {
                // 把它加到当前路径里
                // 包头不包尾
                path.add(s.substring(start, end + 1));
                
                // 继续往后处理
                backtrack(s, end + 1, path, result);
                
                // 回溯：撤销选择，尝试更长的回文
                path.remove(path.size() - 1);
            }
            // 如果不是回文，就直接跳过，不用递归了
        }
    }
    
    /**
     * 判断 s[left...right] 是否是回文串
     */
    private boolean isPalindrome(String s, int left, int right) {
        while (left < right) {
            if (s.charAt(left) != s.charAt(right)) {
                return false;
            }
            left++;
            right--;
        }
        return true;
    }
}
~~~

~~~java
                if (stack.isEmpty() || stack.peek() != '{') {

            } else {
~~~

~~~java
class Solution {
    public String decodeString(String s) {
        Deque<Character> stack = new ArrayDeque<>();
        
        for (char c : s.toCharArray()) {
            if (c != ']') {
                stack.push(c);
            } else {
                // 一旦遇到右括号 ]，说明我们遇到了一个完整的 k[xxx] 结构，需要开始解码这一部分
                // step 1: 取出 [] 内的字符串
                StringBuilder sb = new StringBuilder();
                // Character.isLetter() 判断是不是字母（a-zA-Z）
                // 只把字母放进sb
                while (!stack.isEmpty() && Character.isLetter(stack.peek())) {
                    // 0：把弹出的字符插到 sb 的最前面
                    sb.insert(0, stack.pop());
                }
                String sub = sb.toString();  // [] 内的字符串
                // 跳出循环表示栈顶是 '['
                stack.pop();  // 弹出 '['
                
                // step 2: 获取倍数数字
                sb = new StringBuilder();
                while (!stack.isEmpty() && Character.isDigit(stack.peek())) {
                    sb.insert(0, stack.pop());
                }
                // 把字符串数字转成 int
                int count = Integer.parseInt(sb.toString());  // 倍数
                
                // step 3: 根据倍数把子串再 push 回去
                while (count > 0) {
                    for (char ch : sub.toCharArray()) {
                        stack.push(ch);
                    }
                    count--;
                }
            }
        }
        
        // 把栈里所有字符取出来（逆序插入）
        // 此时栈里面只剩下字母
        StringBuilder ret = new StringBuilder();
        while (!stack.isEmpty()) {
            // 栈是后进先出，所以我们用 insert(0, ...) 来逆序取出
            // 即一直往最前面插入
            ret.insert(0, stack.pop());
        }
        
        return ret.toString();
    }
~~~

~~~java
    public int findKthLargest(int[] nums, int k) {
        // 将 int[] 转为 List<Integer> 方便操作（也可以直接用数组，但 List 更灵活）
        List<Integer> numList = new ArrayList<>();
        for (int num : nums) {
            numList.add(num);
        }
        
        return quickSelect(numList, k);
    }
    
    private int quickSelect(List<Integer> numList, int k) {
        // 随机选择一个基准（pivot）
        // 防止最坏情况（如果数组本来就几乎有序，每次选最左边会导致 O(n²)）
        Random rand = new Random();
        int pivotIndex = rand.nextInt(numList.size());
        // p 就是我们选出来的“基准值”
        int p = numList.get(pivotIndex);
        
        // 三路分区：所有小于 p、等于 p、大于 p的数
        List<Integer> small = new ArrayList<>();
        List<Integer> equal = new ArrayList<>();
        List<Integer> big = new ArrayList<>();
        
        for (int num : numList) {
            if (num < p) {
                small.add(num);
            } else if (num == p) {
                equal.add(num);
            } else {
                big.add(num);
            }
        }
        
        // 第 k 大的元素在 big 中（big 里都是更大的数）
        if (k <= big.size()) {
            return quickSelect(big, k);
        }
        
        // 第 k 大的元素在 equal 中（等于 pivot 的部分）
        // big 里的都是比 p 大的
        // equal 里的都等于 p
        else if (k <=big.size() + equal.size()) {
            return p;
        }else{
            // 第 k 大的元素在 small 中（比 pivot 小的部分）
            // 此时 k 要减去 big 和 equal 的数量
            return quickSelect(small, k - (big.size() + equal.size()));
        }

    }
~~~

~~~java
class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        // 1. 统计频率
        Map<Integer, Integer> freq = new HashMap<>();
        for (int num : nums) {
            freq.put(num, freq.getOrDefault(num, 0) + 1);
        }
        
        // 2. 小顶堆（优先队列）：频率value小的在堆顶
        //   为什么用小顶堆？
        // 因为我们只关心频率最大的 k 个，所以我们让堆里始终保持“当前已知的前 k 高频元素”，
        // 当堆超过 k 个时，就把频率最小的那个扔掉。
        PriorityQueue<Map.Entry<Integer, Integer>> minHeap = 
            new PriorityQueue<>((a, b) -> a.getValue() - b.getValue());
        
        // 3. 只保留 k 个最高频率的元素
        for (Map.Entry<Integer, Integer> entry : freq.entrySet()) {
            minHeap.offer(entry);
            // 如果堆的大小超过了 k，就把堆顶（频率最小的那个）弹出
            // 这样堆里永远只保留当前已知的前 k 高频（堆顶是最小的那个）
            if (minHeap.size() > k) {
                minHeap.poll();  // 弹出频率最小的
            }
        }
        
        // 4. 收集结果
        int[] result = new int[k];
        // 每次弹出的都是当前堆里频率最小的
        // 从后往前放
        // 把最小的频率放 result 最后，把最大的频率放 result 最前面
        for (int i = k - 1; i >= 0; i--) {

            result[i] = minHeap.poll().getKey();
        }
        return result;
    }
}
~~~

~~~java
        int maxReach = 0;  // 初始能到达的最远位置是 0（第 0 个位置）

            maxReach = Math.max(maxReach, i + nums[i]);
        }
        
        // 如果能遍历完整个数组，说明所有位置都能到达，返回 true
        return true;
    }
~~~

~~~java
class Solution {
    public int jump(int[] nums) {
        int maxFar = 0;     // 目前能跳到的最远位置
        int steps = 0;      // 跳跃次数
        int end = 0;        // 上次跳跃能到达的最远边界（下一次起跳的右边界）

        // 注意：循环到 nums.length - 1 就够了，因为最后一个位置不需要再跳
        for (int i = 0; i < nums.length - 1; i++) {
            // 更新目前能跳到的最远距离
            // 只要我能走到 i 这个位置，我就一定能从 i 跳到 i + nums[i]
            maxFar = Math.max(maxFar, i + nums[i]);

            // 如果已经走到了本次跳跃的右边界
            if (i == end) {
                // 把目前能跳到的最远位置作为下一次跳跃的右边界
                // 从当前位置起跳
                end = maxFar;
                // 起跳
                steps++;
            }
        }

        return steps;
    }
}
~~~

~~~java
class Solution {
    // 关键思路：dp[i] = min(dp[i], dp[i - j*j] + 1)
    public int numSquares(int n) {
        // dp[i] 表示凑出数字 i ，最少需要几个完全平方数
        // 0-i一共i+1个位置
        int[] dp = new int[n + 1];
        
        // 初始化：dp[0] = 0，其他先设为最大值（n 个 1 肯定可以）
        // 因为后面要取 min（最小值），所以先给一个“上界”（不可能比 n+1 更大），方便后面更新
        // 为什么填（n+1）？
        // 因为我们要取 min，如果某个位置还没更新过，就保持一个“大值”，后面遇到更好的方案就能把它替换掉
        Arrays.fill(dp, n + 1);  // 比 n 大即可
        dp[0] = 0;
        
        // 遍历每个数字 i
        for (int i = 1; i <= n; i++) {
            // 尝试所有可能的平方数 j*j <= i
            for (int j = 1; j * j <= i; j++) {
                // dp[i] = min(dp[i], dp[i - j*j] + 1)
                // 这里1表示j * j的情况
                dp[i] = Math.min(dp[i], dp[i - j * j] + 1);
            }
        }        
        return dp[n];
    }
~~~

~~~java
class Solution {
    // nums = [3, 5, 6, 2, 7, 1, 8]
    // tails = [1, 5, 6, 7, 8]
    // 这并不是说存在一个序列 1→5→6→7→8（数组里根本没有这个序列）
    // 它真正的含义是：
    // 所有长度为 1 的递增子序列中，最小的结尾是 1
    // 所有长度为 2 的递增子序列中，最小的结尾是 5
    // 所有长度为 3 的递增子序列中，最小的结尾是 6
    // 所有长度为 4 的递增子序列中，最小的结尾是 7
    // 所有长度为 5 的递增子序列中，最小的结尾是 8
    public int lengthOfLIS(int[] nums) {
        if (nums == null || nums.length == 0) {
            return 0;
        }
        
        // tails[i] 表示长度为 i+1 的递增子序列的末尾最小值
        int[] tails = new int[nums.length];
        int len = 0;  // 当前 tails 数组的有效长度
        
        for (int num : nums) {

            // 二分查找：在 tails 中找到第一个 >= num 的位置
            // 因为要用num替换tails里面的值，让tails里面的每一个尾巴更小
            // 如果能找到第一个 >= num 的位置，则替换
            // 如果找不到，说明num最大，直接拼接在tails最后面
            int left = 0, right = len;
            while (left < right) {
                int mid = left + (right - left) / 2;
                // mid 太小了，我要找更大的位置
                if (tails[mid] >= num) {
                    // mid 已经 >= num，可能就是它，或者左边还有更小的
                    right = mid;
                } else {
                    left = mid + 1;
                }
            }
            
            // left 就是我们找到的位置
            // 如果这个位置正好等于当前长度 len，说明 num 比所有现有尾巴都大，可以接在最长序列后面
            if (len == left) {
                len++;
            }
            
            // 不管是延长还是替换，都把这个位置的尾巴更新成 num（因为 num 更小或一样大）
            tails[left] = num;
        }
        
        return len;
    }
~~~

~~~java
class Solution {
~~~

~~~java
        int n = nums.length;
        int sum = 0;
        
        // 1. 计算总和
        for (int num : nums) {
            sum += num;
        }
        
        // 2. 如果总和是奇数，直接返回 false（不可能平分）
        if (sum % 2 != 0) {
            return false;
        }
        
        // 3. 目标是凑出 sum / 2
        int target = sum / 2;
        
        // dp[j] 表示：是否能用前面的数字正好凑出和 j
        boolean[] dp = new boolean[target + 1];
        dp[0] = true;  // 和 0 一定可以（什么都不选）
        
        // 4. 遍历每个数字（物品）
        for (int num : nums) {
            // 如果从前往后遍历，会导致同一个 num 被多次使用
            // 5. 从后往前遍历（0-1 背包经典写法，避免重复使用同一个数字）
            for (int j = target; j >= num; j--) {
                // 如果不拿这个 num 也能凑出 j-num
                // 那拿上这个 num 就能凑出 j
                if (dp[j - num]) {
                    dp[j] = true;
                }
            }
        }
        
        // 最后看能不能凑出 target
        return dp[target];
    }
}
~~~

~~~java
    public int uniquePaths(int m, int n) {
~~~

~~~java
class Solution {

    public int minPathSum(int[][] grid) {
        if (grid == null || grid.length == 0 || grid[0].length == 0) {
            return 0;
        }
        
        int rows = grid.length;
        int cols = grid[0].length;
        
        // dp[j] 表示到达第 i 行、第 j 列的最小路径和
        // 我们只用一行空间（滚动数组）
        int[] dp = new int[cols];
        
        // 初始化第一行：只能一直向右走
        dp[0] = grid[0][0];
        // 第一行只能一直向右走，所以每格的路径和 = 左边一格 + 当前格子值
        for (int j = 1; j < cols; j++) {
            dp[j] = dp[j - 1] + grid[0][j];
        }
        
        // 从第二行开始计算
        for (int i = 1; i < rows; i++) {
            // 第一列只能从上面下来
            dp[0] += grid[i][0];
            
            // 从第二列开始，每一格 = min(上面, 左边) + 当前格子值
            for (int j = 1; j < cols; j++) {
                // dp[j]：上一行的值（从上面下来的路径和
                // dp[j - 1]：当前行左边一格的值（从左边过来的路径和
                dp[j] = Math.min(dp[j], dp[j - 1]) + grid[i][j];
            }
        }
        
        // 最终 dp[cols-1] 就是右下角的最小路径和
        return dp[cols - 1];
    }
~~~

~~~java
class Solution {
    //二维布尔数组 dp[i][j] 表示：
    //字符串 s 从下标 i 到下标 j（包含 i 和 j）这一段是不是回文串
    public String longestPalindrome(String s) {
        int len = s.length();
        // maxLen：记录当前找到的最长回文子串的长度（初始 1，因为单个字符一定是回文）
        int maxLen = 1;
        // begin：记录这个最长回文子串的起始下标（初始 0）
        int begin = 0;
        // dp[i][j] 标识s[i..j]是否是回文串
        boolean[][] dp = new boolean[len][len];
        char[] charArray = s.toCharArray();
        // 从右端点开始遍历
        // 注意从0开始而不是从len-1开始，因为右端点也是慢慢扩大右移的
        for (int j = 0; j < len; j++) {
            // c1 = 当前右端点的字符
            char c1 = charArray[j];
            // 从左端点开始遍历
            for (int i = 0; i <= j; i++) {
                char c2 = charArray[i];
                if (c1 == c2) {
                    // dp[i][j] = (j - i <= 2) || dp[i + 1][j - 1];   
                    // 0/1/2表示单个字符/两个/三个
                    if (j - i <= 2) {
                        dp[i][j] = true;
                        // 中间也是回环
                    } else if (dp[i + 1][j - 1]) {
                        dp[i][j] = true;
                    }
                    if (dp[i][j] && j - i + 1 > maxLen) {
                        maxLen = j - i + 1;
                        // 记录目前最长回文字符串的起始下标
                        begin = i;
                    }
                }
            }
        }
        return s.substring(begin, begin + maxLen);
    }
}
~~~

~~~java
class Solution {
    public void nextPermutation(int[] nums) {
        // 4 5 2 6 3 1 -> 4 5 3 1 2 6
        // 1. 从倒数第二个位置开始，从后往前找降(nums[k] >= nums[k + 1])的位置 k
        // 2. 判断k是否是负数，负数表示整个数组降序，直接全体反转
        // 3. 在 k 的后段找第一个比 nums[k] 大的位置 l
        // 4. 交换 l , k 
        // 5. 把 k 的后段升序




        int n = nums.length;
        
        // 步骤1：从后往前找到第一个“上升对” nums[k] < nums[k+1]
        int k = n - 2;                    // 注意这里是 n-2
        while (k >= 0 && nums[k] >= nums[k + 1]) {
            k--;
        }
        
        // 如果 k == -1，说明整个数组是降序（最大排列）
        if (k == -1) {
            reverse(nums, 0, n - 1);
            return;
        }
        
        // 步骤2：在 [k+1, n-1] 中从后往前找到第一个 > nums[k] 的数（最大的后继）
        int l = n - 1;
        while (l > k && nums[l] <= nums[k]) {
            l--;
        }
        
        // 步骤3：交换 nums[k] 和 nums[l]
        swap(nums, k, l);
        
        // 步骤4：把 [k+1, n-1] 反转成升序（最小排列）
        reverse(nums, k + 1, n - 1);
    }
    
    // 反转数组 [left, right] 区间（包含左右端点）
    private void reverse(int[] nums, int left, int right) {
        while (left < right) {
            swap(nums, left, right);
            left++;
            right--;
        }
    }
    
    // 交换 nums[a] 和 nums[b]
    private void swap(int[] nums, int a, int b) {
        int temp = nums[a];
        nums[a] = nums[b];
        nums[b] = temp;
    }
}
~~~

~~~java
class Solution {
    public int findDuplicate(int[] nums) {
        // 快慢指针（龟兔赛跑）
        int slow = nums[0];
        int fast = nums[0];

        // 第一阶段：快指针走两步，慢指针走一步，找到相遇点
        while (true) {
                slow = nums[slow];           // 慢走一步
                fast = nums[nums[fast]];     // 快走两步
                
                if (slow == fast) {          // 相遇就立刻跳出
                    break;
                }
            }
        // 乌龟和兔子第一次相遇，一定在环里面的某个位置
        // 第二阶段：慢指针回到起点，快指针还留在第一次相遇的位置
        slow = nums[0];
        // 从起点到环入口的距离 = 从相遇点到环入口的距离（沿着环走）
        while (slow != fast) {
            slow = nums[slow];
            fast = nums[fast];
        }

        return slow;
    }
~~~
